## Installation

```bash
npx sketchy-ui add card
```

## Usage

```tsx
import {
  Card,
  CardHeader,
  CardTitle,
  CardDescription,
  CardContent,
  CardFooter,
} from '@/components/ui/card';

<Card>
  <CardHeader>
    <CardTitle>Card Title</CardTitle>
    <CardDescription>Card Description</CardDescription>
  </CardHeader>
  <CardContent>
    <p>Card Content</p>
  </CardContent>
  <CardFooter>
    <p>Card Footer</p>
  </CardFooter>
</Card>
```

## Props

### Card
| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `border` | `'rough' \| 'normal'` | `'rough'` | The type of border drawing. |
